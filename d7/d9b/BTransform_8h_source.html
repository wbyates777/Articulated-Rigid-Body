237 BRBInertia
238 apply( const BRBInertia &rbi ) const
239 // return X^* I X^{-1}
240 {
241 const BMatrix3 ET = arb::transpose(m_E);
242 const BVector3 h = ET * (rbi.h() - (rbi.mass() * m_r));
243 const BMatrix3 rx = arb::cross(m_r);
244 const BMatrix3 aux = (rx * arb::cross(rbi.h())) + ((arb::cross(rbi.h() - rbi.mass() * m_r) * rx));
245 const BMatrix3 I = ET * (rbi.I() + aux) * m_E;
246 return BRBInertia( rbi.mass(), h, I );
247 }

250 BRBInertia
251 applyTranspose( const BRBInertia &rbi ) const
252 // return X^T I X
253 {
254 const BVector3 Eh = m_E * rbi.h();
255 const BVector3 h = Eh + (rbi.mass() * m_r);
256 BMatrix3 I = m_E * rbi.I() * arb::transpose(m_E);
257 BScalar c = arb::dot(Eh + h, m_r);
258 BVector3 col0 = m_r[0] * Eh + h[0] * m_r;
259 BVector3 col1 = m_r[1] * Eh + h[1] * m_r;
260 BVector3 col2 = m_r[2] * Eh + h[2] * m_r;
261 col0[0] -= c; col1[1] -= c; col2[2] -= c;
262 I -= BMatrix3(col0, col1, col2);
263 return BRBInertia( rbi.mass(), h, I );
264 }

266 BABInertia
267 apply( const BABInertia &abi ) const
268 // return X^* I X^{-1}
269 {
270 const BMatrix3 rx = arb::cross(m_r);
271 const BMatrix3 H = abi.H() - (rx * abi.M());
272 const BMatrix3 I = abi.I() - (rx * arb::transpose(abi.H()) ) + (H * rx);
273 const BMatrix3 ET = arb::transpose(m_E);
274 return BABInertia( ET * abi.M() * m_E, ET * H * m_E, ET * I * m_E );
275 }

277 BABInertia
278 applyTranspose( const BABInertia &abi ) const
279 // return X^T I X
280 {
281 const BMatrix3 ET = arb::transpose(m_E);
282 const BMatrix3 M = m_E * abi.M() * ET;
283 const BMatrix3 H = m_E * abi.H() * ET;
284 const BMatrix3 I_orig = m_E * abi.I() * ET;
285 const BMatrix3 rx = arb::cross(m_r);
286 const BMatrix3 rxM = rx * M;
287 const BMatrix3 rxH = rx * H;
288 const BMatrix3 I = I_orig - rxH + (arb::transpose(H) - rxM) * rx;
289 return BABInertia( M, H - arb::transpose(rxM), I );
290 }
