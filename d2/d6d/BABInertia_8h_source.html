293 inline constexpr BMatrix6
294 inverse( const BABInertia &abi )
295 {
296 const BMatrix3 invM = arb::inverse(abi.M());
297 const BMatrix3 T = abi.I() - arb::transpose(abi.H()) * invM * abi.H();
298 const BMatrix3 invT = arb::inverse(T);
299
300 const BMatrix3 topLeft = invT;
301 const BMatrix3 topRight = -invM * abi.H() * invT;
302 const BMatrix3 botLeft = -invT * arb::transpose(abi.H()) * invM;
303 const BMatrix3 botRight = invM + invM * abi.H() * invT * arb::transpose(abi.H()) * invM;
304
305 return BMatrix6(topLeft, topRight, botLeft, botRight);
306 }
