241 inline constexpr BMatrix6
242 inverse( const BRBInertia &I )
243 {
244 assert(I.mass() != 0.0);
245 const BMatrix3 invI(arb::inverse(I.Icom()));
246 const BMatrix3 invM(1.0 / I.mass());
247 const BVector3 com(I.com());
248
249 return BMatrix6( invI, arb::cross(-com) * invI,
250 invI * arb::cross(com), invM + arb::cross(-com) * invI * arb::cross(com) );
251 }
