71 inline BMatrix3
72 exp( const BVector3 &w )
73 {
74 using std::sin;
75 using std::cos;
76
77 const BScalar theta = arb::length(w);
78 const BMatrix3 I(B_IDENTITY_3x3);
79 const BMatrix3 W = arb::cross(w);
80
81 if (theta < 1E-8)
82 {
83 // Rodrigues series near zero: $exp(cross(ω)) ≈ I + cross(ω) + 1/2 cross(ω)^2$
84 return I + W + BScalar(0.5) * (W * W);
85 }
86
87 const BScalar A = sin(theta) / theta;
88 const BScalar B = (1.0 - cos(theta)) / (theta * theta);
89
90 return I + A * W + B * (W * W);
91 }

93 inline BMatrix3
94 leftJacobian( const BVector3 &w )
95 {
96 using std::sin;
97 using std::cos;
98
99 const BScalar theta = arb::length(w);
100 const BMatrix3 I(B_IDENTITY_3x3);
101 const BMatrix3 W = arb::cross(w);
102
103 if (theta < 1E-8)
104 {
105 return I + BScalar(0.5) * W + BScalar(1.0 / 6.0) * (W * W);
106 }
107
108 const BScalar A = (1.0 - cos(theta)) / (theta * theta);
109 const BScalar B = (theta - sin(theta)) / (theta * theta * theta);
110
111 return I + A * W + B * (W * W);
112 }

117 {

138 inline BVector3
139 log( const BMatrix3 &R )
140 {
141 using std::acos;
142 using std::sin;
143
144 const BScalar cos_theta = arb::clamp(0.5 * (arb::trace(R) - 1.0), -1.0, 1.0);
145 const BScalar theta = acos(cos_theta);
146 //assert(!arb::isnan(theta));
147
148 // near identity, use first-order approximation
149 if (theta < 1E-10)
150 {
151 return arb::uncross(R - arb::transpose(R)) * BScalar(0.5);
152 }
153
154 // when theta near M_PI handle rotation singularity
155 if (theta > BScalar(M_PI - 1E-3))
156 {
157 // use the column with the largest diagonal element to ensure numerical stability
158 BVector3 u;
159 if (R[0][0] >= R[1][1] && R[0][0] >= R[2][2])
160 {
161 u = BVector3(R[0][0] + BScalar(1.0), R[1][0], R[2][0]);
162 }
163 else if (R[1][1] >= R[2][2])
164 {
165 u = BVector3(R[0][1], R[1][1] + BScalar(1.0), R[2][1]);
166 }
167 else
168 {
169 u = BVector3(R[0][2], R[1][2], R[2][2] + BScalar(1.0));
170 }
171
172 return arb::normalize(u) * theta;
173 }
174
175 const BMatrix3 W = (R - arb::transpose(R)) * (BScalar(0.5) * theta / sin(theta));
176 return arb::uncross(W);
177 }

179 inline BMatrix3
180 leftJacobianInverse( const BVector3 &w )
181 {
182 using std::tan;
183
184 const BScalar theta = arb::length(w);
185 const BMatrix3 I = B_IDENTITY_3x3;
186 const BMatrix3 W = arb::cross(w);
187
188 if (theta < 1E-8)
189 {
190 return I - BScalar(0.5) * W + BScalar(1.0 / 12.0) * (W * W);
191 }
192
193 const BScalar half_theta = 0.5 * theta;
194 const BScalar cot_half_theta = 1.0 / tan(half_theta);
195
196 const BScalar A = (1.0 - half_theta * cot_half_theta);
197 const BScalar B = BScalar(1.0) / (theta * theta);
198
199 return I - BScalar(0.5) * W + (A * B) * (W * W);
200 }
