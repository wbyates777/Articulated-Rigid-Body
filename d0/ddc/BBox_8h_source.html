210 inline double
211 intersectArea(const BBox& b1, const BBox& b2)
212
213 {
214 double area = 0;
215
216 if (intersect(b1, b2))
217 {
218 glm::dvec3 side(glm::min(b1.top(), b2.top()) - glm::max(b1.bot(), b2.bot()));
219
220 // for 2D case
221 if (side.z == 0.0) side.z = 1.0;
222
223 area = side.x * side.y * side.z;
224 }
225
226 return area;
227 }
